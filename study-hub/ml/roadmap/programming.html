import numpy as np # Create arrays a = np.array([1, 2, 3, 4]) mat = np.zeros((3, 4)) rand = np.random.randn(100, 10) # 100 samples, 10 features # Broadcasting — no loops needed! normalized = (rand - rand.mean(axis=0)) / rand.std(axis=0) # Boolean indexing positives = rand[rand > 0]
