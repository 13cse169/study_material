Start here if you're new to Node.js

What is Node.js?

Node.js is a JavaScript runtime built on Chrome's V8 engine that allows you to run JavaScript on the server side. It uses an event-driven, non-blocking I/O model that makes it lightweight and efficient — perfect for data-intensive real-time applications.

Fast Execution

Powered by V8 engine which compiles JS to native code

Non-Blocking

Async I/O means Node never waits idle

npm Ecosystem

1.8M+ packages available via npm

Single-threaded

One thread handles many concurrent connections

Event-Driven Architecture

Node.js uses an event loop to handle asynchronous operations. When an operation completes (like reading a file or receiving a network request), it emits an event and a callback is triggered.

Non-Blocking I/O

Traditional servers block execution while waiting for I/O. Node.js delegates these operations to the OS and continues executing other code.

blocking.js
// Blocking: waits for each operation to finish
const data1 = readFile('file1.txt'); // Waits...
const data2 = readFile('file2.txt'); // Waits...
console.log(data1, data2);        // Only runs after both
// Total time = time1 + time2
non-blocking.js
// Non-Blocking: both start simultaneously
Promise.all([
  fs.readFile('file1.txt'),
  fs.readFile('file2.txt')
]).then(([data1, data2]) => {
  console.log(data1, data2);
});
// Total time ≈ max(time1, time2) 🚀

JavaScript Fundamentals

Closures

A closure is a function that retains access to its outer (lexical) scope even after the outer function has returned. They are fundamental to Node.js module patterns and callbacks.

Closures
function createCounter() {
  let count = 0; // Private variable!
  return {
    increment: () => ++count,
    decrement: () => --count,
    getCount:  () => count
  };
}

const counter = createCounter();
counter.increment(); // 1
counter.increment(); // 2
counter.getCount();  // 2 — 'count' is private

Promises vs Async/Await

Callback Hell 😱
getUser(userId, (err, user) => {
  if (err) return console.log(err);
  getOrders(user.id, (err, orders) => {
    if (err) return console.log(err);
    getDetails(orders[0].id, (err, detail) => {
      if (err) return console.log(err);
      console.log(detail); // Pyramid of doom!
    });
  });
});
Promises ⛓️
getUser(userId)
  .then(user => getOrders(user.id))
  .then(orders => getDetails(orders[0].id))
  .then(detail => console.log(detail))
  .catch(err => console.log(err));
// Much cleaner, but still chained

// Parallel execution:
Promise.all([fetch1(), fetch2(), fetch3()])
  .then(([r1, r2, r3]) => console.log(r1, r2, r3));
Async/Await ✨
async function loadUserOrder(userId) {
  try {
    const user    = await getUser(userId);
    const orders  = await getOrders(user.id);
    const detail  = await getDetails(orders[0].id);
    return detail; // Reads like synchronous code!
  } catch (err) {
    console.error('Error:', err);
  }
}

Build a Todo REST API

Todo API — Full CRUD
const express = require('express');
const app = express();
app.use(express.json());

let todos = [];
let nextId = 1;

// GET all todos
app.get('/todos', (req, res) =>
  res.json(todos));

// GET one todo
app.get('/todos/:id', (req, res) => {
  const todo = todos.find(t => t.id === +req.params.id);
  todo ? res.json(todo) : res.status(404).json({ error: 'Not found' });
});

// POST create todo
app.post('/todos', (req, res) => {
  const todo = { id: nextId++, text: req.body.text, done: false };
  todos.push(todo);
  res.status(201).json(todo);
});

// PUT update todo
app.put('/todos/:id', (req, res) => {
  const idx = todos.findIndex(t => t.id === +req.params.id);
  if (idx === -1) return res.status(404).json({ error: 'Not found' });
  todos[idx] = { ...todos[idx], ...req.body };
  res.json(todos[idx]);
});

// DELETE todo
app.delete('/todos/:id', (req, res) => {
  todos = todos.filter(t => t.id !== +req.params.id);
  res.status(204).send();
});

app.listen(3000);

Authentication & Security
JWT, bcrypt, security best practices

JWT Authentication Flow

1. User sends credentials

POST /login with email + password

2. Server verifies & signs JWT

jwt.sign(payload, secret, options)

3. Client stores token

Usually in memory or httpOnly cookie

4. Client sends token in header

Authorization: Bearer <token>

5. Server verifies & extracts payload

jwt.verify(token, secret)

JWT + bcrypt Auth System
const bcrypt = require('bcryptjs');
const jwt    = require('jsonwebtoken');

// SIGNUP
async function signup(req, res) {
  const { email, password } = req.body;
  const hash = await bcrypt.hash(password, 12); // Cost factor: 12
  const user = await User.create({ email, password: hash });
  res.status(201).json({ message: 'Account created!' });
}

// LOGIN
async function login(req, res) {
  const { email, password } = req.body;
  const user = await User.findOne({ email });
  if (!user) return res.status(401).json({ error: 'Invalid credentials' });

  const match = await bcrypt.compare(password, user.password);
  if (!match) return res.status(401).json({ error: 'Invalid credentials' });

  const token = jwt.sign(
    { userId: user._id, role: user.role },
    process.env.JWT_SECRET,
    { expiresIn: '7d' }
  );
  res.json({ token });
}

// AUTH MIDDLEWARE
function protect(req, res, next) {
  const token = req.headers.authorization?.split(' ')[1];
  if (!token) return res.status(401).json({ error: 'No token' });
  try {
    req.user = jwt.verify(token, process.env.JWT_SECRET);
    next();
  } catch { res.status(401).json({ error: 'Invalid token' }); }
}

Event Loop — Deep Dive

The event loop processes callbacks in a specific order across multiple phases. Understanding this order is crucial for writing performant Node.js code.

Event Loop Order
console.log('1: Sync');
setTimeout(() => console.log('5: setTimeout'), 0);
Promise.resolve().then(() => console.log('3: Promise microtask'));
queueMicrotask(() => console.log('4: queueMicrotask'));
setImmediate(() => console.log('6: setImmediate'));
console.log('2: Sync');
// Order: 1 → 2 → 3 → 4 → 5 → 6
// Microtasks (Promises) always run BEFORE macrotasks (setTimeout)

Streams

Streams process data in chunks instead of loading it all into memory — essential for handling large files and HTTP responses.

Streams & Pipe
const fs   = require('fs');
const zlib = require('zlib');

// Stream a 2GB file without loading it into memory!
fs.createReadStream('bigfile.txt')
  .pipe(zlib.createGzip())    // Transform: compress
  .pipe(fs.createWriteStream('bigfile.gz'));

// Custom Transform stream
const { Transform } = require('stream');
const upper = new Transform({
  transform(chunk, encoding, callback) {
    this.push(chunk.toString().toUpperCase());
    callback();
  }
});

process.stdin.pipe(upper).pipe(process.stdout);

Worker Threads

For CPU-intensive tasks that would block the event loop, offload them to Worker Threads.

Worker Threads
// main.js
const { Worker } = require('worker_threads');

function runWorker(data) {
  return new Promise((resolve, reject) => {
    const worker = new Worker('./worker.js', { workerData: data });
    worker.on('message', resolve);
    worker.on('error', reject);
  });
}
const result = await runWorker({ n: 42 });

Clustering

Cluster Module
const cluster = require('cluster');
const os      = require('os');

if (cluster.isPrimary) {
  const cpuCount = os.cpus().length;
  console.log(`Forking ${cpuCount} workers...`);

  for (let i = 0; i < cpuCount; i++) cluster.fork();

  cluster.on('exit', (worker) => {

REST Best Practices

  • Use nouns, not verbs: /users not /getUsers
  • Use plural resource names: /articles not /article
  • Use HTTP methods correctly: GET (read), POST (create), PUT/PATCH (update), DELETE (delete)
  • Return consistent error shapes: {"error":"message","code":"ERR_CODE"}
  • Version your API: /api/v1/users
  • Paginate list endpoints: ?page=1&limit=20

Real-Time Systems
WebSockets, Socket.io, Chat App

WebSockets vs HTTP

HTTP is request-response: client asks, server responds. WebSockets are bidirectional: the server can push data to clients at any time — perfect for chat, live dashboards, and multiplayer games.

Socket.io Chat App

server.js
const express = require('express');
const http    = require('http');
const { Server } = require('socket.io');

const app    = express();
const server = http.createServer(app);
const io     = new Server(server, {
  cors: { origin: '*' }
});

io.on('connection', (socket) => {
  console.log(`User connected: ${socket.id}`);

  socket.on('join_room', (room) => {
    socket.join(room);
    socket.to(room).emit('user_joined', socket.id);
  });

  socket.on('send_message', ({ room, message }) => {
    io.to(room).emit('receive_message', {
      author: socket.id,
      message,
      timestamp: new Date()
    });
  });

  socket.on('disconnect', () => {
    console.log(`User disconnected: ${socket.id}`);
  });
});

server.listen(3000);
